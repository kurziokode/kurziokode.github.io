Phase 4: Backend API Logic

Ο κώδικας που ελέγχει τα limits, το AI και τις πληρωμές.

📁 app/api/ai-action/route.ts 
import { GoogleGenerativeAI } from "@google/generative-ai";
import { adminDb, adminAuth } from "@/lib/firebase-admin";
import { NextResponse } from "next/server";
import { FieldValue } from "firebase-admin/firestore";

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);
const FREE_LIMIT = 5;

export async function POST(req: Request) {
  try {
    // 1. SECURITY: Verify Token
    const authHeader = req.headers.get("Authorization");
    if (!authHeader?.startsWith("Bearer ")) return NextResponse.json({ error: "Unauthorized" }, { status: 401 });
    const token = authHeader.split("Bearer ")[1];
    const decodedToken = await adminAuth.verifyIdToken(token);
    const userId = decodedToken.uid;

    const { prompt, contextData, action } = await req.json();

    // 2. DB CHECK
    const userRef = adminDb.collection("users").doc(userId);
    const userSnap = await userRef.get();

    if (!userSnap.exists) {
      await userRef.set({ isPro: false, usageCount: 0, lastUsageDate: new Date().toISOString() });
    }

    const userData = userSnap.exists ? userSnap.data() : { isPro: false, usageCount: 0 };
    
    // 3. RESET COUNTER (Daily)
    const lastDate = new Date(userData?.lastUsageDate);
    const today = new Date();
    if (lastDate.getDate() !== today.getDate()) {
       await userRef.update({ usageCount: 0, lastUsageDate: today.toISOString() });
       if(userData) userData.usageCount = 0;
    }

    // 4. FEATURE GATING
    if (action === 'PREMIUM_FEATURE' && !userData?.isPro) {
        return NextResponse.json({ error: "PREMIUM_ONLY" }, { status: 403 });
    }
    if (!userData?.isPro && (userData?.usageCount || 0) >= FREE_LIMIT) {
      return NextResponse.json({ error: "LIMIT_REACHED" }, { status: 403 });
    }

    // 5. CALL AI
    const modelName = userData?.isPro ? "gemini-1.5-pro" : "gemini-1.5-flash";
    const model = genAI.getGenerativeModel({ model: modelName });
    const result = await model.generateContent(`User: ${prompt} \n Context: ${JSON.stringify(contextData)}`);
    const text = result.response.text();

    // 6. INCREMENT
    if (!userData?.isPro) {
      await userRef.update({ usageCount: FieldValue.increment(1) });
    }

    return NextResponse.json({ result: text, isPro: userData?.isPro });

  } catch (error) {
    return NextResponse.json({ error: "Internal Error" }, { status: 500 });
  }
}
📁 app/api/webhooks/stripe/route.ts 
import { headers } from "next/headers";
import { NextResponse } from "next/server";
import Stripe from "stripe";
import { adminDb } from "@/lib/firebase-admin";

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: "2023-10-16" });
const endpointSecret = process.env.STRIPE_WEBHOOK_SECRET!;

export async function POST(req: Request) {
  const body = await req.text();
  const signature = headers().get("stripe-signature") as string;
  let event: Stripe.Event;

  try {
    event = stripe.webhooks.constructEvent(body, signature, endpointSecret);
  } catch (err: any) {
    return NextResponse.json({ error: err.message }, { status: 400 });
  }

  if (event.type === "checkout.session.completed") {
    const session = event.data.object as Stripe.Checkout.Session;
    const userId = session.metadata?.userId;
    if (userId) {
      await adminDb.collection("users").doc(userId).update({ isPro: true, plan: "premium" });
    }
  }
  return NextResponse.json({ received: true });
}

Phase 6: Security Rules

🔥 Firestore Rules
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read: if request.auth != null && request.auth.uid == userId;
      allow write: if false; // Only Admin SDK writes here!
    }
  }
}
